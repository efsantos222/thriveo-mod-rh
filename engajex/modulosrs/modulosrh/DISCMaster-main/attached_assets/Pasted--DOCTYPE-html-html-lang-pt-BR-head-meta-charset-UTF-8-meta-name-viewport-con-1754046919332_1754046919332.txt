<script>
    let processedData = [];
    let originalData = [];

    // Configuração do drag and drop
    const uploadSection = document.getElementById('uploadSection');
    const fileInput = document.getElementById('fileInput');

    uploadSection.addEventListener('dragover', (e) => {
        e.preventDefault();
        uploadSection.classList.add('dragover');
    });

    uploadSection.addEventListener('dragleave', () => {
        uploadSection.classList.remove('dragover');
    });

    uploadSection.addEventListener('drop', (e) => {
        e.preventDefault();
        uploadSection.classList.remove('dragover');
        const files = e.dataTransfer.files;
        if (files.length > 0) {
            handleFile(files[0]);
        }
    });

    fileInput.addEventListener('change', (e) => {
        if (e.target.files.length > 0) {
            handleFile(e.target.files[0]);
        }
    });

    function showMessage(message, type) {
        const errorMsg = document.getElementById('errorMessage');
        const successMsg = document.getElementById('successMessage');
        
        if (type === 'error') {
            errorMsg.textContent = message;
            errorMsg.style.display = 'block';
            successMsg.style.display = 'none';
        } else {
            successMsg.textContent = message;
            successMsg.style.display = 'block';
            errorMsg.style.display = 'none';
        }
    }

    function hideMessages() {
        document.getElementById('errorMessage').style.display = 'none';
        document.getElementById('successMessage').style.display = 'none';
    }

    function showLoading() {
        document.getElementById('loading').style.display = 'block';
    }

    function hideLoading() {
        document.getElementById('loading').style.display = 'none';
    }

    async function handleFile(file) {
        if (!file.name.match(/\.(xlsx|xls)$/)) {
            showMessage('Por favor, selecione um arquivo Excel (.xlsx ou .xls)', 'error');
            return;
        }

        hideMessages();
        showLoading();

        try {
            const data = await readExcelFile(file);
            originalData = data;
            processedData = calculateDISC(data);
            displayResults();
            showMessage(`Arquivo processado com sucesso! ${processedData.length} pessoas analisadas.`, 'success');
        } catch (error) {
            showMessage('Erro ao processar arquivo: ' + error.message, 'error');
            console.error('Erro:', error);
        } finally {
            hideLoading();
        }
    }

    function readExcelFile(file) {
        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const data = new Uint8Array(e.target.result);
                    const workbook = XLSX.read(data, { type: 'array' });
                    const firstSheetName = workbook.SheetNames[0];
                    const worksheet = workbook.Sheets[firstSheetName];
                    const jsonData = XLSX.utils.sheet_to_json(worksheet);
                    
                    validateData(jsonData);
                    resolve(jsonData);
                } catch (error) {
                    reject(error);
                }
            };
            reader.onerror = () => reject(new Error('Erro ao ler o arquivo'));
            reader.readAsArrayBuffer(file);
        });
    }

    function validateData(data) {
        if (!data || data.length === 0) {
            throw new Error('Planilha vazia ou sem dados');
        }

        const firstRow = data[0];
        const keys = Object.keys(firstRow);
        
        if (keys.length < 30) {
            throw new Error(`Planilha deve ter pelo menos 30 colunas (ID, Nome + 28 questões). Encontradas: ${keys.length}`);
        }

        // Verificar se tem colunas básicas
        const hasId = keys.some(key => key.toLowerCase().includes('id'));
        const hasName = keys.some(key => key.toLowerCase().includes('nome') || key.toLowerCase().includes('name'));
        
        if (!hasId || !hasName) {
            throw new Error('Planilha deve conter colunas "ID" e "Nome"');
        }

        // Verificar se os valores estão no range correto
        for (let row of data.slice(0, Math.min(5, data.length))) {
            const values = Object.values(row).slice(2); // Pular ID e Nome
            for (let i = 0; i < Math.min(28, values.length); i++) {
                const val = Number(values[i]);
                if (isNaN(val) || val < 1 || val > 5) {
                    console.warn(`Valor fora do range (1-5) encontrado: ${values[i]} na linha ${data.indexOf(row) + 1}`);
                }
            }
        }
    }

    function calculateDISC(data) {
        return data.map(row => {
            const keys = Object.keys(row);
            const id = row[keys[0]];
            const name = row[keys[1]];
            
            // Extrair os 28 valores (7 para cada perfil)
            const values = [];
            for (let i = 2; i < Math.min(30, keys.length); i++) {
                const val = Number(row[keys[i]]);
                values.push(isNaN(val) ? 3 : Math.max(1, Math.min(5, val))); // Default 3, clamp entre 1-5
            }

            // Preencher com valores padrão se necessário
            while (values.length < 28) {
                values.push(3);
            }

            // Calcular pontuações para cada perfil
            const profiles = {
                D: values.slice(0, 7).reduce((sum, val) => sum + val, 0),
                I: values.slice(7, 14).reduce((sum, val) => sum + val, 0),
                S: values.slice(14, 21).reduce((sum, val) => sum + val, 0),
                C: values.slice(21, 28).reduce((sum, val) => sum + val, 0)
            };

            // Encontrar perfil dominante
            const dominant = Object.keys(profiles).reduce((a, b) => profiles[a] > profiles[b] ? a : b);

            // Calcular percentuais
            const total = Object.values(profiles).reduce((sum, val) => sum + val, 0);
            const percentages = {};
            Object.keys(profiles).forEach(key => {
                percentages[key] = Math.round((profiles[key] / total) * 100);
            });

            return {
                id,
                name,
                profiles,
                percentages,
                dominant,
                total,
                rawValues: values
            };
        });
    }

    function displayResults() {
        document.getElementById('resultsSection').style.display = 'block';
        
        // Mostrar estatísticas gerais
        displayStats();
        
        // Mostrar resultados individuais
        displayIndividualResults();
    }

    function displayStats() {
        const statsGrid = document.getElementById('statsGrid');
        const totalPeople = processedData.length;
        
        // Contar perfis dominantes
        const dominantCounts = { D: 0, I: 0, S: 0, C: 0 };
        processedData.forEach(person => {
            dominantCounts[person.dominant]++;
        });

        // Calcular médias
        const averages = { D: 0, I: 0, S: 0, C: 0 };
        processedData.forEach(person => {
            Object.keys(averages).forEach(profile => {
                averages[profile] += person.percentages[profile];
            });
        });
        Object.keys(averages).forEach(profile => {
            averages[profile] = Math.round(averages[profile] / totalPeople);
        });

        statsGrid.innerHTML = `
            <div class="stat-item">
                <div class="stat-number">${totalPeople}</div>
                <div class="stat-label">Total de Pessoas</div>
            </div>
            <div class="stat-item">
                <div class="stat-number">${dominantCounts.D}</div>
                <div class="stat-label">Perfil D Dominante</div>
            </div>
            <div class="stat-item">
                <div class="stat-number">${dominantCounts.I}</div>
                <div class="stat-label">Perfil I Dominante</div>
            </div>
            <div class="stat-item">
                <div class="stat-number">${dominantCounts.S}</div>
                <div class="stat-label">Perfil S Dominante</div>
            </div>
            <div class="stat-item">
                <div class="stat-number">${dominantCounts.C}</div>
                <div class="stat-label">Perfil C Dominante</div>
            </div>
            <div class="stat-item">
                <div class="stat-number">${averages.D}%</div>
                <div class="stat-label">Média Perfil D</div>
            </div>
            <div class="stat-item">
                <div class="stat-number">${averages.I}%</div>
                <div class="stat-label">Média Perfil I</div>
            </div>
            <div class="stat-item">
                <div class="stat-number">${averages.S}%</div>
                <div class="stat-label">Média Perfil S</div>
            </div>
            <div class="stat-item">
                <div class="stat-number">${averages.C}%</div>
                <div class="stat-label">Média Perfil C</div>
            </div>
        `;
    }

    function displayIndividualResults() {
        const resultsGrid = document.getElementById('resultsGrid');
        
        resultsGrid.innerHTML = processedData.map(person => `

    function getProfileName(letter) {
        const names = {
            'D': 'Dominância',
            'I': 'Influência', 
            'S': 'Estabilidade',
            'C': 'Conformidade'
        };
        return names[letter] || letter;
    }

    function exportToExcel() {
        if (processedData.length === 0) {
            showMessage('Nenhum dado para exportar', 'error');
            return;
        }

        const exportData = processedData.map(person => ({
            'ID': person.id,
            'Nome': person.name,
            'D (%)': person.percentages.D,
            'I (%)': person.percentages.I,
            'S (%)': person.percentages.S,
            'C (%)': person.percentages.C,
            'D (Pontos)': person.profiles.D,
            'I (Pontos)': person.profiles.I,
            'S (Pontos)': person.profiles.S,
            'C (Pontos)': person.profiles.C,
            'Perfil Dominante': person.dominant,
            'Nome Perfil Dominante': getProfileName(person.dominant),
            'Total Pontos': person.total
        }));

        const ws = XLSX.utils.json_to_sheet(exportData);
        const wb = XLSX.utils.book_new();
        XLSX.utils.book_append_sheet(wb, ws, "Resultados DISC");
        
        const filename = `resultados_disc_${new Date().toISOString().split('T')[0]}.xlsx`;
        XLSX.writeFile(wb, filename);
        
        showMessage('Arquivo Excel exportado com sucesso!', 'success');
    }

    function exportToPDF() {
        showMessage('Função de exportação PDF será implementada em breve', 'error');
    }
</script>
